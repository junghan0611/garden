```shell
sudo apt install curl

sudo curl -fsSLo /usr/share/keyrings/brave-browser-archive-keyring.gpg https://brave-browser-apt-release.s3.brave.com/brave-browser-archive-keyring.gpg

echo "deb [signed-by=/usr/share/keyrings/brave-browser-archive-keyring.gpg] https://brave-browser-apt-release.s3.brave.com/ stable main"|sudo tee /etc/apt/sources.list.d/brave-browser-release.list

sudo apt update

sudo apt install brave-browser
```

```bash
  flatpak install flathub engineer.atlas.Nyxt
  flatpak install flathub org.qutebrowser.qutebrowser
  flatpak run engineer.atlas.Nyxt
  flatpak run org.qutebrowser.qutebrowser
```

아래와 같이 설치할 필요 없을 듯.

```text

$ flatpak --user remote-add --if-not-exists flathub https://flathub.org/repo/flathub.flatpakrepo
$ flatpak --user install --noninteractive org.gnome.Sdk/x86_64/44
$ flatpak --user install ~/Downloads/nyxt-3.6.1.flatpak
$ flatpak run engineer.atlas.Nyxt
```
