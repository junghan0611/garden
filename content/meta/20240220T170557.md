```text
# Regolith 3.1 - 2014-02-19

echo deb "[arch=amd64 signed-by=/usr/share/keyrings/regolith-archive-keyring.gpg] https://regolith-desktop.org/testing-ubuntu-lunar-amd64 lunar main" | sudo tee /etc/apt/sources.list.d/regolith.list

sudo apt update
sudo apt upgrade

# sudo vi /etc/regolith/i3/config : common-wm -> common
# sudo vi /etc/regolith/sway/config : common-wm -> common

sudo rm -Rf /usr/share/regolith/*
sudo rm -Rf /usr/share/i3xrocks/*

# 내 설정을 그대로 사용하면 된다.
```

#### Install {#install}

```text
$ wget -qO - https://regolith-desktop.org/regolith.key | \
gpg --dearmor | sudo tee /usr/share/keyrings/regolith-archive-keyring.gpg > /dev/null

$ echo deb "[arch=amd64 signed-by=/usr/share/keyrings/regolith-archive-keyring.gpg] \
https://regolith-desktop.org/release-3_0-ubuntu-lunar-amd64 lunar main" | \
sudo tee /etc/apt/sources.list.d/regolith.list

```

```text

# 2023-10-15 add alt key
## Navigate // Relative Window // <> k j h l ##
set_from_resource $wm.binding.left wm.binding.left h
set_from_resource $wm.binding.right wm.binding.right l
set_from_resource $wm.binding.up wm.binding.up k
set_from_resource $wm.binding.down wm.binding.down j
bindsym $mod+$alt+$wm.binding.left focus left
bindsym $mod+$alt+$wm.binding.down focus down
bindsym $mod+$alt+$wm.binding.up focus up
bindsym $mod+$alt+$wm.binding.right focus right

# 2023-10-15 add alt key
## Navigate // Workspaces 1-10 // <> 0..9 ##
bindsym $mod+$alt+$ws1_key workspace number $ws1
bindsym $mod+$alt+$ws2_key workspace number $ws2
bindsym $mod+$alt+$ws3_key workspace number $ws3
bindsym $mod+$alt+$ws4_key workspace number $ws4
bindsym $mod+$alt+$ws5_key workspace number $ws5
bindsym $mod+$alt+$ws6_key workspace number $ws6
bindsym $mod+$alt+$ws7_key workspace number $ws7
bindsym $mod+$alt+$ws8_key workspace number $ws8
```

-   기본 설정
    -   super + 0-9 : i3 workspace : 기본
    -   super + meta + left/right : i3 prev/next workspace
    -   M-0 ~ M-9 : winum-select-window -- Spacemacs 기본 세팅

-   수정 설정
    -   s + hjkl : emacs windows
    -   super + ctrl + 0-9 : i3 ws11-19 인데 안쓴다. emacs tab-bar 로 변경
    -   super + ctrl + hl : i3 ws11-19 인데 안쓴다. emacs tab-bar 로 변경
    -   super + meta + hjkl : 기본 설정 없음. i3 창 이동
    -   super + meta + 0-9 : i3 워크스페이스 옴김 -&gt; i3 창 이동
