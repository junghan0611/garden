  (defun command-of-the-day ()
    "Show the documentation for a random command."
    (interactive)
    (let ((commands))
      (mapatoms (lambda (s)
                  (when (commandp s) (push s commands))))
      (describe-function
       (nth (random (length commands)) commands))))
